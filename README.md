# Running Procedure pf WTKO-CNN Pipeline:

conda install -c bioconda bedtools

Download WTKO_CNN Folder and unzip it.

Then Run: 

Install mm10.fa or any other genome as per requirement


python WTKO_CNN.py --wt.bed WT.bed --ko.bed KO.bed --genome mm10.fa --outdir Results

Internally, the pipeline will do the following operations:


# Get the WT and KO Specific Peaks

WT-specific regions

bedtools intersect -v -a wt.bed -b ko.bed > wtspecific.bed

KO-specific regions

bedtools intersect -v -a ko.bed -b wt.bed > kospecific.bed


# Generate 600 bp Peak files and extract the sequences
bed to fasta.py
# cnn_attention.py
CNN Architecture with attention layer to classify WT vs KO Peaks.
# test_attention.py
Testing on held-out Chromosome using Best model. 
# motifattention.py
Helps to discriminative k-mer extraction and subsequently clustered using agglomerative clustering to group similar sequence patterns.

It will generate 
| Output                                 | Description                                                                               |
| -------------------------------------- | ----------------------------------------------------------------------------------------- |
| Saliency_TopPositions_Flat.csv         | Flattened saliency file with one row per discriminative k-mer.                            |
| Motif_Consensus.csv                    | Consensus sequence and number of k-mers for each cluster.                                 |                                       
| Cluster_FASTA                          | Folder containing one FASTA file per cluster (`cluster_0.fasta`, `cluster_1.fasta`, ...). |
| Motif_Logos                            | Folder containing sequence logo PNGs for every cluster.                                   |


# Apply MEME on each cluster generated by Agglomerative clustering

Meme command : for i in {1..10}
do
    meme cluster_${i}.fasta \
        -dna \
        -oc cluster_${i}_output \
        -nmotifs 5 \
        -minw 6 \
        -maxw 12 \
        -mod zoops
done
# Apply MEME and  TOMTOM Command to report statistically significant similarities based on Pearson correlation on all clusters

for file in *.fasta
do
    name=$(basename "$file" .fasta)

    meme "$file" \
        -dna \
        -oc "${name}_output" \
        -nmotifs 5 \
        -minw 6 \
        -maxw 12 \
        -mod zoops

    tomtom \
        -oc "${name}_tomtom" \
        -dist pearson \
        -thresh 0.5 \
        "${name}_output/meme.txt" \
        /mnt/c/Jasper/combined.meme
done
